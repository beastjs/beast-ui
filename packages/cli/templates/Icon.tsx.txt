'use client'

import { useId } from 'react'
import { ICON_VIEWBOX, iconMarkup } from './icons'
import type { IconProps } from './types'

export default function Icon({ name, size = 20, color, label, style, ...props }: IconProps) {
  const instanceId = useId()
  return (
    <svg
      xmlns="http://www.w3.org/2000/svg"
      viewBox={ICON_VIEWBOX}
      width={size}
      height={size}
      fill="currentColor"
      strokeLinecap="round"
      strokeLinejoin="round"
      {...props}
      role={label ? 'img' : undefined}
      aria-label={label}
      aria-hidden={label ? undefined : true}
      style={{ flexShrink: 0, color, ...style }}
      dangerouslySetInnerHTML={{ __html: iconMarkup(name, instanceId) }}
    />
  )
}
